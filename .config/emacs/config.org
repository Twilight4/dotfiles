;;;  -*- lexical-binding: t; -*-

(defvar elpaca-installer-version 0.12)
(defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
(defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
(defvar elpaca-sources-directory (expand-file-name "sources/" elpaca-directory))
(defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                              :ref nil :depth 1 :inherit ignore
                              :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                              :build (:not elpaca-activate)))
(let* ((repo  (expand-file-name "elpaca/" elpaca-sources-directory))
       (build (expand-file-name "elpaca/" elpaca-builds-directory))
       (order (cdr elpaca-order))
       (default-directory repo))
  (add-to-list 'load-path (if (file-exists-p build) build repo))
  (unless (file-exists-p repo)
    (make-directory repo t)
    (when (<= emacs-major-version 28) (require 'subr-x))
    (condition-case-unless-debug err
        (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                  ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                  ,@(when-let* ((depth (plist-get order :depth)))
                                                      (list (format "--depth=%d" depth) "--no-single-branch"))
                                                  ,(plist-get order :repo) ,repo))))
                  ((zerop (call-process "git" nil buffer t "checkout"
                                        (or (plist-get order :ref) "--"))))
                  (emacs (concat invocation-directory invocation-name))
                  ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                        "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                  ((require 'elpaca))
                  ((elpaca-generate-autoloads "elpaca" repo)))
            (progn (message "%s" (buffer-string)) (kill-buffer buffer))
          (error "%s" (with-current-buffer buffer (buffer-string))))
      ((error) (warn "%s" err) (delete-directory repo 'recursive))))
  (unless (require 'elpaca-autoloads nil t)
    (require 'elpaca)
    (elpaca-generate-autoloads "elpaca" repo)
    (let ((load-source-file-function nil)) (load "./elpaca-autoloads"))))
(add-hook 'after-init-hook #'elpaca-process-queues)
(elpaca `(,@elpaca-order))

;; Install use-package support
(elpaca elpaca-use-package
  ;; Enable :elpaca use-package keyword.
  (elpaca-use-package-mode)
  ;; Assume :elpaca t unless otherwise specified.
  (setq use-package-always-ensure t))

;; Block until current queue processed.
(elpaca-wait)

* GUI Configuration
** Frame configuration
These settings only take effect on graphical frames — text-terminal frames ignore font =:family=/=:height= — so a single =config.org= serves both =emacsclient -c= (GUI) and =emacsclient -nw= (terminal) frames from the same =--daemon=. The font is =JetBrainsMonoNerdFont= at 14pt — the ligature-capable variant (not the "NL" no-ligature cut kitty still uses) so programming ligatures (arrows, comparison/logic operators) render via the =ligature.el= block below. To match kitty exactly, point kitty at the same variant later. 

#+begin_src emacs-lisp
;; Font — JetBrains Mono Nerd Font (ligature variant) @ 14pt.
;; Switched from the "NL" (no-ligature) cut so programming ligatures compose;
;; `default-frame-alist' is the daemon-safe path: the :font frame parameter is
;; applied to each frame as it is created, so it reaches a pgtk GUI frame even
;; though no frame exists yet when the daemon loads config.
(set-face-attribute 'default nil
                    :family "JetBrainsMono Nerd Font"
                    :height 140)
(add-to-list 'default-frame-alist '(font . "JetBrainsMono Nerd Font-14"))

     "**" "***" "*>" "#{" "#[" "#:" "#=" "#_("))
  (global-ligature-mode t))

** Transparency
It's always better to set transparency in option for respective applications that supports it rather than setting it in window manager.

#+begin_src emacs-lisp
;; `alpha-background' (Emacs 29+): only the background goes translucent — text
;; and glyphs stay opaque — unlike the legacy `alpha' parameter which faded
;; everything. Range: 0 (fully transparent) .. 100 (opaque). 91 is the default
;; translucent level — wallpaper reads through, text stays crisp. TTY frames
;; ignore `alpha-background', so every command here is a harmless no-op there.
(defvar tl/frame-alpha-background 91
  "Current background alpha (0-100). 100=opaque, lower=more translucent.
Single source of truth: every transparency command reads and updates this,
then mirrors it to all live frames and `default-frame-alist' (new frames).")
(defvar tl/frame-alpha-saved 91
  "Last translucent (<100) alpha. `tl/toggle-frame-transparency' restores it.
Updated whenever a non-100 alpha is set, so toggle-off→on returns to the
level the user last chose rather than always 91.")
(defvar tl/frame-alpha-default 91
  "Default translucent alpha. `tl/reset-frame-transparency' returns here.")

(defun tl/set-frame-transparency (alpha)
  "Set background alpha to ALPHA (0-100) on all live and future frames.
ALPHA is clamped to [0,100]. Values below 100 are remembered in
`tl/frame-alpha-saved' so the toggle can restore them. Interactive
prefix reads a number from the minibuffer."
  (interactive "NAlpha (0-100, lower=more transparent): ")
  (let ((alpha (max 0 (min 100 alpha))))
    (setq tl/frame-alpha-background alpha)
    (when (< alpha 100)
      (setq tl/frame-alpha-saved alpha))
    (dolist (f (frame-list))
      (set-frame-parameter f 'alpha-background alpha))
    (setq default-frame-alist
          (cons (cons 'alpha-background alpha)
                (assq-delete-all 'alpha-background default-frame-alist)))
    (message "Frame alpha: %d%%" alpha)))

(defun tl/toggle-frame-transparency ()
  "Toggle between opaque (100) and the last translucent alpha.
Off (100) → restore `tl/frame-alpha-saved'. On (<100) → go to 100."
  (interactive)
  (if (= tl/frame-alpha-background 100)
      (tl/set-frame-transparency tl/frame-alpha-saved)
    (tl/set-frame-transparency 100)))

(defun tl/more-transparent (n)
  "Decrease alpha by N (default 5) — more see-through.
Mirrors `text-scale-decrease' (C--) under the transparency prefix."
  (interactive "p")
  (tl/set-frame-transparency (- tl/frame-alpha-background (if (> n 1) n 5))))

(defun tl/less-transparent (n)
  "Increase alpha by N (default 5) — more opaque.
Mirrors `text-scale-increase' (C-=) under the transparency prefix."
  (interactive "p")
  (tl/set-frame-transparency (+ tl/frame-alpha-background (if (> n 1) n 5))))

(defun tl/reset-frame-transparency ()
  "Reset alpha to `tl/frame-alpha-default' (91)."
  (interactive)
  (tl/set-frame-transparency tl/frame-alpha-default))

;; Default transparency for new frames — `tl/set-frame-transparency' keeps this
;; in sync on every change.
(add-to-list 'default-frame-alist
             (cons 'alpha-background tl/frame-alpha-background))

;; Bindings live under the `C-x C-t' "Toggle/Table" prefix (tl/leader-keys
;; block, search "Toggle/Table"), as `C-t t/-/=/0/s' — which-key lists them.
#+end_src

** Tab bar & workspaces
=tab-bar= — one tab per workspace/context, plus =tabspaces= for buffer isolation and session save/restore across restarts. Named workspaces are built from recipes via =tl/open-workspace= (bound under =C-x w=).

#+begin_src emacs-lisp
;; Native tab bar = workspace tabs. `tab-bar-show' hides the bar until a
;; second tab exists.
;; M-N selects tab N (M-9 = last, M-0 = most-recent). Must precede
;; `tab-bar-mode': enabling it calls `tab-bar--define-keys', which reads this
;; and binds M-digits in `tab-bar-mode-map' — overriding `digit-argument'.
(setq tab-bar-select-tab-modifiers '(meta))
(tab-bar-mode 1)
(tab-bar-history-mode 1)
(setq tab-bar-show 1
      tab-bar-new-tab-choice "*dashboard*"
      ;; Only the tabs themselves — no `«»' history arrows (left) or `+'
      tab-bar-format '(tab-bar-format-tabs)
      ;; Tab width = its name length (kitty-style). The default `t' stretches
      ;; every tab to fill the bar
      tab-bar-auto-width nil
      ;; One-char gap between tabs, rendered in the `tab-bar' face (= the bar
      ;; background), so it reads as a #040305 gap between #16161e tab blocks.
      tab-bar-separator " "
      ;; No per-tab `x' close button; closing is via keys (M-C / C-x w d).
      tab-bar-close-button-show nil)

;; Kitty-style tab names: pad one space each side and carry the tab face onto
;; the padding, so the highlight doesn't hug the text — the name floats on a
;; small colored margin like kitty's tab padding.
(defun tl/tab-name-kitty (tab _i)
  "Return TAB's name padded with the tab face (kitty-style, no text-hugging)."
  (let ((name (alist-get 'name tab))
        (face (if (eq (car tab) 'current-tab)
                  'tab-bar-tab 'tab-bar-tab-inactive)))
    (propertize (concat " " name " ") 'face face)))
(setq tab-bar-tab-name-format-function #'tl/tab-name-kitty)

;; --- Mirror kitty's tab-bar + cursor colors. The pywal-generated
;; colors-kitty.conf is the source of truth (theme.conf includes it LAST, so it
;; overrides); `tl/sync-kitty-tab-colors' re-reads it and applies the colors,
;; so regenerating the pywal scheme is mirrored here with no second edit.
;; Defined here; applied after `load-theme' in "Color theme" and hooked into
;; `enable-theme-functions' to survive theme reloads. ---
(defconst tl/kitty-theme-file
  (expand-file-name "~/.cache/wal/colors-kitty.conf")
  "Pywal-generated kitty color file, included LAST by kitty's theme.conf.
Holds the real tab/cursor colors (active/inactive tab fg+bg, cursor).
tab_bar_background is not set here — it stays #040305 from theme.conf.")

(defun tl/kitty-color (key &optional fallback)
  "Read the #rrggbb color for KEY (e.g. \"active_tab_foreground\") from kitty.
Return the hex string, or FALLBACK when colors-kitty.conf or the key is absent."
  (or (and (file-readable-p tl/kitty-theme-file)
           (with-temp-buffer
             (insert-file-contents tl/kitty-theme-file)
             (goto-char (point-min))
             (when (re-search-forward
                    (concat "^\\s-*" (regexp-quote key)
                            "\\s-+\\(#[0-9a-fA-F]\\{6\\}\\)")
                    nil t)
               (match-string 1))))
      fallback))

(defun tl/sync-kitty-tab-colors (&optional _theme)
  "Apply kitty's tab-bar and cursor colors to the Emacs faces.
Sources: `tl/kitty-theme-file' (pywal) — active/inactive tab fg+bg, cursor."
  (interactive)
  (let ((bar-bg (tl/kitty-color "tab_bar_background"      "#040305"))
        (act-fg (tl/kitty-color "active_tab_foreground"   "#061C24"))
        (act-bg (tl/kitty-color "active_tab_background"   "#adb6b7"))
        (ina-fg (tl/kitty-color "inactive_tab_foreground" "#adb6b7"))
        (ina-bg (tl/kitty-color "inactive_tab_background" "#061C24"))
        (cursor (tl/kitty-color "cursor"                  "#adb6b7")))
    (set-face-attribute 'tab-bar nil :background bar-bg :box nil)
    (set-face-attribute 'tab-bar-tab nil
                        :background act-bg :foreground act-fg :box nil)
    (set-face-attribute 'tab-bar-tab-inactive nil
                        :background ina-bg :foreground ina-fg :box nil)
    (set-face-attribute 'cursor nil :background cursor)
    (set-cursor-color cursor)))

;; --- Named workspace recipes (mirror the kitty session files) -------------
;; Each entry: (WORKSPACE-SYM (TAB-NAME . OPENER) ...). OPENER is a command
;; symbol or a zero-arg function, run inside the tab it creates.
(defconst tl/workspace-recipes
  '((agenda
     ("Agenda"     . tl/org-agenda-o)
     ("New Org"    . tl/buffer-org-new-kitty)
     ("Cheatsheet" . (lambda () (find-file "~/desktop/workspace/dotfiles/.config/emacs/emacs-cheatsheet.org")))
     ("Calendar"   . (lambda () (org-goto-calendar) (delete-other-windows)))
     ("Journal"    . org-roam-dailies-goto-today))
    (work
     ("IDE"         . (lambda () (treemacs)))
     ("cli" . (lambda () (tl/ghostel-new-named "ghostel-cli")))
     ("omp-agents"  . tl/agent-shell-omp-start)
     ("gemini"      . tl/gptel-open-gemini)))
  "Alist of WORKSPACE -> list of (TAB-NAME . OPENER) recipes.")

(defun tl/--tab-index (name)
  "1-based index of the tab named NAME, or nil if absent."
  (cl-loop for tab in (tab-bar-tabs)
           for i from 1
           when (equal (alist-get 'name tab) name) return i))

(defun tl/open-workspace (name)
  "Open workspace NAME, creating its recipe tabs if absent.
Existing tabs are reused (idempotent), so re-opening just focuses the
workspace. Switches to the workspace's first tab when done."
  (interactive
   (list (intern (completing-read "Workspace: "
                                  (mapcar #'car tl/workspace-recipes)))))
  (let ((recipe (alist-get name tl/workspace-recipes)))
    (if (not recipe)
        (user-error "No recipe for workspace `%s'" name)
      (dolist (entry recipe)
        (pcase-let ((`(,tab-name . ,opener) entry))
          (let ((idx (tl/--tab-index tab-name)))
            (if idx
                (tab-bar-select-tab idx)
              (tab-bar-new-tab)
              (tab-bar-rename-tab tab-name)
              (condition-case err
                  (if (commandp opener) (call-interactively opener) (funcall opener))
                (error (message "opener %S failed in tab %S: %s"
                                opener tab-name (error-message-string err))))))))
      (let ((first (car (mapcar #'car recipe))))
        (when-let ((idx (tl/--tab-index first)))
          (tab-bar-select-tab idx))))))

;; --- Cycle focus across buffers visible in EVERY tab (consult) ---------
;; Each tab carries a frozen `ws' (window-state) snapshot; only the live tab
;; has real windows, so other tabs' buffers are recovered by walking `ws' for
;; every `(buffer . NAME)' leaf (stable regardless of how internal windows
;; encode children/siblings across Emacs builds). `consult' lists them with
;; live preview + a tab annotation; choosing one switches to its tab and
;; raises the window that displays it. Bound to `C-x w b'.
(defun tl/window-state-buffers (ws)
  "Return buffer names displayed in window-state WS."
  (let (names)
    (cl-labels ((walk (x)
                  (cond
                   ((and (consp x) (eq (car x) 'buffer))
                    (let ((nm (or (and (stringp (cdr x)) (cdr x))   ; (buffer . "n") dotted
                                  (and (stringp (cadr x)) (cadr x))))) ; (buffer "n" ...) list form
                      (when nm (push nm names))))
                   ((consp x) (walk (car x)) (walk (cdr x))))))
      (walk ws))
    (delete-dups (nreverse names))))

(defun tl/switch-through-visible-tab-buffer ()
  "Jump with `consult' to a buffer visible in any `tab-bar' tab.
Cycle candidates with `C-n'/`C-p'/<up>/<down>; `RET' switches to the
owning tab and raises the window displaying the chosen buffer."
  (interactive)
  (unless (bound-and-true-p tab-bar-mode)
    (user-error "Enable `tab-bar-mode' first (M-x tab-bar-mode)"))
  (let* ((tabs (tab-bar-tabs))
         (table (make-hash-table :test 'equal)) ; buf -> tab-names
         (order nil))                          ; first-seen buf order
    (dolist (tab tabs)
      (let* ((tname (alist-get 'name tab))
             ;; The current tab keeps its live state in the frame, not a `ws'
             ;; snapshot (a snapshot is captured only when you LEAVE a tab), so
             ;; snapshot the live frame; other tabs use their stored `ws'.
             (ws (or (alist-get 'ws tab)
                     (when (eq (car tab) 'current-tab)
                       (window-state-get nil t)))))
        (dolist (b (tl/window-state-buffers ws))
          (when (get-buffer b)
            (unless (gethash b table) (push b order))
            (cl-pushnew tname (gethash b table) :test #'equal)))))
    (setq order (nreverse order))
    (if (null order)
        (user-error "No visible buffers in any tab")
      (let* ((orig-win (selected-window))
             (orig-buf (window-buffer orig-win))
             (choice
              (consult--read
               order
               :prompt "Visible buffer (all tabs): "
               :category 'buffer
               :require-match t
               :annotate (lambda (cand)
                           (if-let ((ts (gethash cand table)))
                               (format " [%s]" (mapconcat #'identity ts ", "))
                             ""))
               :state (consult--buffer-state))))
        ;; consult previews/returns by switching the ORIGINATING window to
        ;; `choice' (`consult--buffer-action' = switch-to-buffer); undo that so
        ;; the window we ran from is left untouched — we jump to the buffer's
        ;; own tab/window instead. norecord avoids disturbing the buffer list
        ;; (consult restores it on `return'). Do this BEFORE the tab switch,
        ;; while orig-win is still a live window.
        (unless (eq (window-buffer orig-win) orig-buf)
          (with-selected-window orig-win
            (switch-to-buffer orig-buf t)))
        (let* ((tab-name (car (gethash choice table)))

              (select-window win)

;; pgtk GUI Emacs bridges the kill-ring and the Wayland clipboard two ways:
;;  - `interprogram-cut-function' (= `gui-select-text') leaks every kill
;;    (C-u/C-o/C-w/M-d) to the clipboard;
;;  - `interprogram-paste-function' (= `gui-selection-value') makes
;;    `current-kill' *re-import* the clipboard on every call, so any text the
;;    user copied outside Emacs (or an old error string) gets prepended to the
;;    kill-ring, clobbering freshly-killed subtrees. This broke `org-paste-
;;    subtree' (every agenda DONE/clarify/refile clobbered the just-copied
;;    subtree with stale clipboard text → "The kill is not a (set of) tree(s)").
;; Disable BOTH so kills stay in the kill-ring; explicit `tl/*-to-clipboard'
;; wrappers (which call wl-copy directly) and `wl-paste' remain the only path
;; to/from the system clipboard.
(setq interprogram-cut-function nil
      interprogram-paste-function nil)

;; With the GUI clipboard bridge off (above), `C-y'/`yank' see only the
;; kill-ring in EVERY buffer, so text copied outside Emacs (browser, etc.)
;; never pastes.  Re-enabling `interprogram-paste-function' would reopen
;; the 2026-07-19 org-mode loop (every `current-kill' re-imports and
;; clobbers freshly-killed subtrees).  Instead, pull the Wayland clipboard
;; into the kill-ring on `focus-in' -- the moment you return to Emacs after
;; copying.  Editing kills stay local; unchanged clipboards are skipped, so

** Word motion
=tl/subword-mode-toggle= cycles word motion globally (bound to =M-`=) through three mutually exclusive states: default → subword → superword → default. Subword makes =C-f=/=C-b= stop inside CamelCase identifiers (=camelCaseWord= → =camel= / =Case= / =Word=); superword treats symbols like =snake_case_word= as a single word. This pairs with =(global-subword-mode 0)= in [[Better defaults]] — Emacs starts in the default state, and =M-`= flips between the two behaviors without ever leaving both on at once.

#+begin_src emacs-lisp
(defun tl/subword-mode-toggle ()
  "Cycle the global word-motion mode: default -> subword -> superword -> default.
Subword: `C-f'/`C-b' stop inside CamelCase identifiers.
Superword: they treat symbols like snake_case as a single word.
The two modes are mutually exclusive; only one is active at a time."
  (interactive)
  (cond
   ;; default -> subword
   ((and (not global-subword-mode) (not global-superword-mode))
    (global-superword-mode -1)
    (global-subword-mode 1)
    (message "Subword mode"))
   ;; subword -> superword
   (global-subword-mode
    (global-subword-mode -1)
    (global-superword-mode 1)
    (message "Superword mode"))
   ;; superword -> default
   (t
    (global-subword-mode -1)
    (global-superword-mode -1)
    (message "Word motion: default"))))

** Code Folding
The built-in =outline-minor-mode= provides structured code folding in modes such as Emacs Lisp and Python, allowing users to collapse and expand sections based on headings or indentation levels. This feature enhances navigation and improves the management of large files with hierarchical structures.

Alternatively, =hs-minor-mode= offers basic code folding for blocks defined by curly braces, functions, or other language-specific delimiters. However, for more flexible folding that supports multiple nested levels, =outline-minor-mode= is generally the preferred choice, as it enables finer control over section visibility in deeply structured code.

For example, to enable =outline-minor-mode= in Emacs Lisp:
#+begin_src emacs-lisp
;; The built-in outline-minor-mode provides structured code folding in modes
;; such as Emacs Lisp and Python, allowing users to collapse and expand sections
;; based on headings or indentation levels. This feature enhances navigation and
;; improves the management of large files with hierarchical structures.
(use-package outline
  :ensure nil
  :commands outline-minor-mode
  :hook
  ((emacs-lisp-mode . outline-minor-mode)
   ;; Use " ▼" instead of the default ellipsis "..." for folded text to make
   ;; folds more visually distinctive and readable.
   (outline-minor-mode
    .
    (lambda()
      (let* ((display-table (or buffer-display-table (make-display-table)))
             (face-offset (* (face-id 'shadow) (ash 1 22)))
             (value (vconcat (mapcar (lambda (c) (+ face-offset c)) " ▼"))))
        (set-display-table-slot display-table 'selective-display value)
        (setq buffer-display-table display-table))))))
#+end_src

* Editing Features
** Improved file management
Function =delete-this-file= deletes the current file and kills the buffer associated with it.
It first checks if there is a file being edited in the buffer.
If not, it throws an error.
Then, it prompts the user for confirmation to delete the file. If the user confirms, it proceeds to delete the file using =delete-file= and kills the buffer using =kill-this-buffer=.

Function =rename-this-file-and-buffer= renames both the current buffer and the file it’s visiting to a new name specified by the user.
It takes user input for the new name using the interactive keyword, checks if the buffer is visiting a file, renames the file if it exists, updates the visited file name, and renames the buffer accordingly.

#+begin_src emacs-lisp
(defun delete-this-file ()
  "Delete the current file, and kill the buffer."
  (interactive)
  (unless (buffer-file-name)
    (error "No file is currently being edited"))
  (when (yes-or-no-p (format "Delete file '%s'?"
                             (file-name-nondirectory buffer-file-name)))
    (delete-file (buffer-file-name))
    (kill-this-buffer)))

(defun rename-this-file-and-buffer (new-name)
  "Renames both current buffer and file it's visiting to NEW-NAME."
  (interactive "sNew name: ")
  (let ((name (buffer-name))
        (filename (buffer-file-name)))
    (unless filename
      (error "Buffer '%s' is not visiting a file!" name))
    (progn
      (when (file-exists-p filename)
        (rename-file filename new-name 1))
      (set-visited-file-name new-name)
      (rename-buffer new-name))))
#+end_src

(defun unpackaged/org-return-dwim (&optional default)
  "A helpful replacement for `org-return-indent'.  With prefix, call `org-return-indent'.

On headings, move point to position after entry content.  In
lists, insert a new item or end the list, with checkbox if
appropriate.  In tables, insert a new row or end the table."
  ;; Inspired by John Kitchin: http://kitchingroup.cheme.cmu.edu/blog/2017/04/09/A-better-return-in-org-mode/
  (interactive "P")
  (if default
      (org-return t)
    (cond
     ((eq 'link (car (org-element-context)))
      ;; Link: Open it.
      (org-open-at-point-global))

     ((org-at-heading-p)
      ;; Heading: Move to position after entry content.
      ;; NOTE: This is probably the most interesting feature of this function.
      (let ((heading-start (org-entry-beginning-position)))
        (goto-char (org-entry-end-position))
        (cond ((and (org-at-heading-p)
                    (= heading-start (org-entry-beginning-position)))
               ;; Entry ends on its heading; add newline after
               (end-of-line)
               (insert "\n\n"))
              (t
               ;; Entry ends after its heading; back up
               (forward-line -1)
               (end-of-line)
               (when (org-at-heading-p)
                 ;; At the same heading
                 (forward-line)
                 (insert "\n")
                 (forward-line -1))
               (while (not (looking-back "\\(?:[[:blank:]]?\n\\)\\{3\\}" nil))
                 (insert "\n"))
               (forward-line -1)))))

     ((org-at-item-checkbox-p)
      ;; Checkbox: Insert new item with checkbox.
      (org-insert-todo-heading nil))

     ((org-in-item-p)
      ;; Plain list.  Yes, this gets a little complicated...
      (let ((context (org-element-context)))
        (if (or (eq 'plain-list (car context))  ; First item in list
                (and (eq 'item (car context))
                     (not (eq (org-element-property :contents-begin context)
                              (org-element-property :contents-end context))))
                (unpackaged/org-element-descendant-of 'item context))  ; Element in list item, e.g. a link
            ;; Non-empty item: Add new item.
            (org-insert-item)
          ;; Empty item: Close the list.
          (delete-region (line-beginning-position) (line-end-position))
          (insert "\n"))))

     ((when (fboundp 'org-inlinetask-in-task-p)
        (org-inlinetask-in-task-p))
      ;; Inline task: Don't insert a new heading.
      (org-return t))

     ((org-at-table-p)
      (cond ((save-excursion
               (beginning-of-line)
               ;; See `org-table-next-field'.
               (cl-loop with end = (line-end-position)
                        for cell = (org-element-table-cell-parser)
                        always (equal (org-element-property :contents-begin cell)
                                      (org-element-property :contents-end cell))
                        while (re-search-forward "|" end t)))
             ;; Empty row: end the table.
             (delete-region (line-beginning-position) (line-end-position))
             (org-return t))
            (t
             ;; Non-empty row: call `org-return-indent'.
             (org-return t))))
     (t
      ;; All other cases: call `org-return-indent'.
	  (org-return t)))))

** Session persistence & crash recovery
Two layers protect against a daemon/frame exit (PGTK Emacs dies when the Wayland display drops): =super-save= writes file-visiting buffers to disk on focus/buffer/idle events (so unsaved edits survive), and =desktop-save-mode= snapshots the full buffer/window/point layout and restores it on restart. Versioned backups stay OFF (single-user, single-daemon). Both write under the =no-littering= =user-emacs-directory= (~/.cache/emacs/).

#+begin_src emacs-lisp
;; Save file-visiting buffers to the real file on natural events — the
;; replacement for the disabled `auto-save-default'. Files are always on disk.
(use-package super-save
  :ensure t
  :demand t
  :config
  (setq super-save-auto-save-when-idle t
        super-save-idle-duration 5
        super-save-silent t)
  (super-save-mode 1))

;; Snapshot the full session (buffers, windows, point) and restore on restart.
;; Save on KILL, not an idle timer: `desktop-auto-save-timeout' is an idle timer
;; (fires only after N s idle following a window-config change)
;; `desktop-save-mode' puts its real saver (`desktop-kill') on
;; `kill-emacs-query-functions' = interactive C-x C-c only; `(kill-emacs)' and a
;; daemon stop run only `kill-emacs-hook', whose desktop handler just releases
;; the lock. So we add an explicit save to `kill-emacs-hook' to cover every exit.
;; Caveat: a HARD crash (SIGKILL / Wayland drop on PGTK) runs no hook at all —
;; then you restore the last clean snapshot; `super-save' keeps file contents
;; safe either way, so only the window LAYOUT is at risk.
(desktop-save-mode 1)
;; `desktop.el' installs an anonymous `after-init-hook' lambda that calls
;; `desktop-read' whenever `desktop-save-mode' is non-nil.  In a frameless
;; daemon that buffer/frame restore aborts startup before `server-start'
;; ("Error: server did not start correctly"), so neutralize the auto-restore
;; in daemon mode only: clear the mode var BEFORE desktop's lambda runs
;; (prepend) so it skips the read, then re-enable it AFTER (append) so the
;; save side keeps working.  Plain GUI startup is unchanged.
(when (daemonp)
  (add-hook 'after-init-hook (lambda () (setq desktop-save-mode nil)) nil)
  (add-hook 'after-init-hook (lambda () (desktop-save-mode 1)) t))
(setq desktop-save t                        ; save on exit without prompting
      desktop-auto-save-timeout nil         ; disable the idle autosave timer
      desktop-restore-frames t
      desktop-load-locked-desktop nil       ; avoid dual-daemon restore race
      desktop-path (list (expand-file-name "var/desktop" user-emacs-directory))
      desktop-files-not-to-save
      "\\(^/[^/:]*:\\)\\|\\(sitemap\\.org\\)") ; skip TRAMP + transient
;; Don't persist registers across restarts — leave register state session-local.
(setq desktop-globals-to-save (delq 'register-alist desktop-globals-to-save))
;; Guarantee a snapshot on EVERY exit path (interactive + `(kill-emacs)' + stop).
(defun tl/desktop-save-on-kill ()
  "Save the desktop on any kill path, including non-interactive `kill-emacs'."
  (when (and desktop-save-mode desktop-dirname)
    (let ((desktop-save t))          ; save unconditionally, no prompt
      (desktop-save desktop-dirname))))
(add-hook 'kill-emacs-hook #'tl/desktop-save-on-kill)
;; Restore on FIRST client frame (daemon), not at after-init. The daemonp
;; guard above disables after-init `desktop-read' (it crashes the frameless
;; daemon's `server-start'); this takes over once a real display exists, so
;; `desktop-read' is safe and gets a frame to lay buffers/windows into.
;; Restores: file buffers (with point/mark, in list order) + the saved
;; frameset (window splits, tab-bar tabs, frame set). Runs once per daemon
;; life; a TTY first-frame defers to a later GUI frame (`display-graphic-p').
(defvar tl/desktop-restored nil
  "Non-nil once `desktop-read' has restored the session into the daemon.")
(defun tl/desktop-restore-on-first-frame ()
  "Restore the saved desktop the first time a GUI client frame is created."
  (unless tl/desktop-restored
    (when (and desktop-save-mode (display-graphic-p))
      (let ((file (expand-file-name desktop-base-file-name (car desktop-path))))
        (when (file-exists-p file)
          (setq tl/desktop-restored t)
          (desktop-read (car desktop-path)))))))
(add-hook 'server-after-make-frame-hook #'tl/desktop-restore-on-first-frame)
#+end_src

** Swap window states
=window-swap-states= (built-in, =window.el=) swaps two windows' full state — buffer, point, mark, start, and hscroll — so unlike =ace-swap-window= (which exchanges only buffers and drops your reading position) nothing is lost. With exactly two windows it swaps the selected window with the other one directly, no picker — the reliable common case; with three or more it prompts via the =ace-window= picker (=aw-select=, dispatch disabled) when available, falling back to the next window.

#+begin_src emacs-lisp
(defun tl/swap-window ()
  "Swap the selected window's state with another window.
Uses `window-swap-states', so buffer, point, mark, and start all
travel together (unlike `ace-swap-window', which swaps only buffers).
With exactly two windows, swaps with the other one directly -- no
picker, the reliable common case. With three or more, prompts for the
target with the `ace-window' picker (`aw-select') when available,
falling back to the next window otherwise."
  (interactive)
  (let* ((selected (selected-window))
         (target (if (= (count-windows) 2)
                     (next-window selected)
                   (if (fboundp 'aw-select)
                       (let ((aw-dispatch-alist nil)) ; picker only, no dispatch actions
                         (aw-select "Swap with window: "))
                     (next-window selected)))))
    (if (and (window-live-p target) (not (eq target selected)))
        (progn (window-swap-states selected target)
               (select-window target))
      (message "No window to swap with"))))
#+end_src

** Cycle window position
=window-swap-states= cycles the selected window's buffer one step through =window-list= (wrapping) — the analogue of kitty's =move_window_forward= / =move_window_backward=. Unlike [[#buffer-move][Buffer-move]] (directional: left/right) and [[#swap-window-states][Swap window states]] (ace-picker pick), this moves the window one position forward/backward in window order. Full state travels (buffer, point, mark), and focus follows the buffer.

#+begin_src emacs-lisp
(defun tl/move-window-forward ()
  "Move the selected window's buffer to the next window (kitty move_window_forward).
Swaps full window state with the next window in `window-list',
wrapping, and follows the buffer.  No-op with a single window."
  (interactive)
  (let* ((wins (window-list))
         (target (or (cadr (memq (selected-window) wins)) (car wins))))
    (window-swap-states nil target)
    (select-window target)))

(defun tl/move-window-backward ()
  "Move the selected window's buffer to the previous window (kitty move_window_backward).
Swaps full window state with the previous window in `window-list',
wrapping, and follows the buffer.  No-op with a single window."
  (interactive)
  (let* ((wins (reverse (window-list)))
         (target (or (cadr (memq (selected-window) wins)) (car wins))))
    (window-swap-states nil target)
    (select-window target)))

** Marginalia
Marginalia is nice, but the file metadata annotations are a little too plain.
Specifically, I have these gripes:
- File attributes would be nicer if coloured
- I don’t care about the user/group information if the user/group is me
- When a file time is recent, a relative age (e.g. 2h ago) is more useful than the date
- An indication of file fatness would be nice

Thanks to the =marginalia-annotator-registry=, we don’t have to advise, we can just add a new =file= annotator.
Another small thing is the face used for docstrings. At the moment it’s (italic shadow), but I don’t like that.

;; Treemacs-only zoxide travel.  This reads zoxide's db.zo directly instead of
;; calling the zoxide executable, so invoking it never starts a shell process or

(with-eval-after-load 'treemacs
  (require 'cl-lib)

  (defun tl/treemacs--set-workspace-projects (workspace projects)
    "Set WORKSPACE's Treemacs project list to PROJECTS.
Set the struct slot by offset at runtime.  This defun is macroexpanded
before Treemacs registers its types, so a `setf' of `cl-struct-slot-value'
\(or the `treemacs-workspace->projects' setter) errors at load time on an
unknown `treemacs-workspace' type; resolving the slot offset at call time
sidesteps that load-time type check."
    (let ((idx (cl-struct-slot-offset 'treemacs-workspace 'projects)))
      (if (eq (cl-struct-sequence-type 'treemacs-workspace) 'list)
          (setcar (nthcdr idx workspace) projects)
        (aset workspace idx projects))))

  (defun tl/zoxide--db-file ()
    "Return the zoxide database path without invoking zoxide."
    (let ((data-dir (or (getenv "_ZO_DATA_DIR")
                        (expand-file-name
                         "zoxide"
                         (or (getenv "XDG_DATA_HOME")
                             (expand-file-name "~/.local/share"))))))
      (unless (file-name-absolute-p data-dir)
        (user-error "_ZO_DATA_DIR must be absolute: %s" data-dir))
      (expand-file-name "db.zo" (file-name-as-directory data-dir))))

  (defun tl/zoxide--read-db-bytes ()
    "Read zoxide's binary database as a unibyte string."
    (let ((db-file (tl/zoxide--db-file)))
      (unless (file-readable-p db-file)
        (user-error "No readable zoxide database at %s" db-file))
      (with-temp-buffer
        (set-buffer-multibyte nil)
        (let ((coding-system-for-read 'binary))
          (insert-file-contents-literally db-file))
        (buffer-string))))

  (defun tl/zoxide--ensure-bytes (bytes offset size)
    "Signal if BYTES cannot provide SIZE bytes at OFFSET."
    (when (> (+ offset size) (length bytes))
      (user-error "Corrupt zoxide database: truncated record")))

  (defun tl/zoxide--u32-le (bytes offset)
    "Read an unsigned little-endian u32 from BYTES at OFFSET."
    (tl/zoxide--ensure-bytes bytes offset 4)
    (+ (aref bytes offset)
       (ash (aref bytes (+ offset 1)) 8)
       (ash (aref bytes (+ offset 2)) 16)
       (ash (aref bytes (+ offset 3)) 24)))

  (defun tl/zoxide--u64-le (bytes offset)
    "Read an unsigned little-endian u64 from BYTES at OFFSET."
    (tl/zoxide--ensure-bytes bytes offset 8)
    (let ((value 0)
          (shift 0))
      (dotimes (i 8 value)
        (setq value (+ value (ash (aref bytes (+ offset i)) shift))
              shift (+ shift 8)))))

  (defun tl/zoxide--f64-le (bytes offset)
    "Read an IEEE-754 little-endian f64 from BYTES at OFFSET."
    (let* ((bits (tl/zoxide--u64-le bytes offset))
           (negative (not (zerop (logand (ash bits -63) 1))))
           (exponent (logand (ash bits -52) #x7ff))
           (fraction (logand bits (1- (ash 1 52))))
           (sign (if negative -1.0 1.0))
           (fraction-scale (float (ash 1 52))))
      (cond
       ((= exponent 0)
        (* sign (expt 2.0 -1022) (/ fraction fraction-scale)))
       ((= exponent #x7ff)
        0.0)
       (t
        (* sign
           (expt 2.0 (- exponent 1023))
           (+ 1.0 (/ fraction fraction-scale)))))))

  (defun tl/zoxide--score (rank last-accessed)
    "Return zoxide's frecency score for RANK and LAST-ACCESSED."
    (let ((age (max 0 (- (floor (float-time)) last-accessed))))
      (* rank
         (cond
          ((< age 3600) 4.0)
          ((< age 86400) 2.0)
          ((< age 604800) 0.5)
          (t 0.25)))))

  (defun tl/zoxide--read-entry (bytes offset)
    "Read one zoxide entry from BYTES at OFFSET.
Return (ENTRY . NEXT-OFFSET), where ENTRY is (PATH RANK LAST-ACCESSED SCORE)."
    (let* ((path-size (tl/zoxide--u64-le bytes offset))
           (path-start (+ offset 8))
           (path-end (+ path-start path-size)))
      (tl/zoxide--ensure-bytes bytes path-start path-size)
      (let* ((path (decode-coding-string
                    (substring bytes path-start path-end)
                    'utf-8-unix))
             (rank-offset path-end)
             (last-accessed-offset (+ rank-offset 8))
             (next-offset (+ last-accessed-offset 8))
             (rank (tl/zoxide--f64-le bytes rank-offset))
             (last-accessed (tl/zoxide--u64-le bytes last-accessed-offset)))
        (cons (list path rank last-accessed (tl/zoxide--score rank last-accessed))
              next-offset))))

  (defun tl/zoxide--entries ()
    "Return zoxide database entries sorted by zoxide frecency."
    (let* ((bytes (tl/zoxide--read-db-bytes))
           (version (tl/zoxide--u32-le bytes 0))
           (count (tl/zoxide--u64-le bytes 4))
           (offset 12)
           entries)
      (unless (= version 3)
        (user-error "Unsupported zoxide database version %s; expected 3" version))
      (when (> count 100000)
        (user-error "Refusing suspicious zoxide database with %s entries" count))
      (dotimes (_ count)
        (let ((entry+offset (tl/zoxide--read-entry bytes offset)))
          (push (car entry+offset) entries)
          (setq offset (cdr entry+offset))))
      (sort entries
            (lambda (a b)
              (let ((score-a (nth 3 a))
                    (score-b (nth 3 b)))
                (if (= score-a score-b)
                    (string-lessp (car a) (car b))
                  (> score-a score-b)))))))

  (defun tl/zoxide--directory-candidates ()
    "Return existing zoxide directories in frecency order."
    (let ((seen (make-hash-table :test 'equal))
          candidates)
      (dolist (entry (tl/zoxide--entries) (nreverse candidates))
        (let ((path (car entry)))
          (when (and (file-directory-p path)
                     (not (gethash path seen)))
            (puthash path t seen)
            (push path candidates))))))

  (defun tl/treemacs-zoxide-travel ()
    "Show and expand a zoxide-ranked directory as the focused Treemacs root.
This command is intentionally valid only inside `treemacs-mode'.  It reads the
zoxide database directly, prompts with `completing-read', updates the current
Treemacs buffer, then expands the new root as `treemacs-TAB-action' would,
without calling zoxide, `find-file', `dired', `switch-to-buffer', or Treemacs
window-selection commands."
    (interactive)
    (unless (eq major-mode 'treemacs-mode)
      (user-error "Run tl/treemacs-zoxide-travel from a Treemacs buffer"))
    (let* ((candidates (tl/zoxide--directory-candidates))
           (selection (progn
                        (unless candidates
                          (user-error "No existing directories found in zoxide database"))
                        (completing-read "Treemacs zoxide: " candidates nil t)))
           (path (treemacs-canonical-path selection))
           (name (file-name-nondirectory (directory-file-name path)))
           (workspace (treemacs-current-workspace)))
      (tl/treemacs--set-workspace-projects
       workspace
       (list (treemacs-project->create!
              :name name
              :path path
              :path-status (treemacs--get-path-status path))))
      (treemacs--invalidate-buffer-project-cache)
      (treemacs--consolidate-projects)
      (goto-char (point-min))
      (when-let ((button (treemacs-current-button)))
        (when (eq (treemacs-button-get button :state) 'root-node-closed)
          (treemacs-TAB-action)))
      (treemacs--evade-image)
      (hl-line-highlight)
      (when-let ((window (get-buffer-window (current-buffer) (selected-frame))))
        (set-window-point window (point)))
      path))

(defun tl/gptel-suffix-send-with-kill-ring-clipboard (args)
  "Send with ARGS and mirror gptel kill-ring output to the Wayland clipboard.
The `kill-new' advice is installed only while gptel's asynchronous callback
runs, so failures that never write to the kill-ring cannot leave stale advice
behind."
  (unless (require 'gptel nil t)
    (user-error "gptel is not available"))
  (letrec ((request-advice
            (lambda (orig-fn prompt &rest request-args)
              (let* ((callback (plist-get request-args :callback))
                     (request-args (copy-sequence request-args)))
                (when (functionp callback)
                  (setq request-args
                        (plist-put
                         request-args :callback
                         (lambda (resp info &optional raw)
                           (letrec ((copy-advice
                                     (lambda (&rest _)
                                       (when (car-safe kill-ring)
                                         (tl/wl-copy (car kill-ring))
                                         (message "gptel response copied to kill-ring and clipboard.")))))
                             (advice-add 'kill-new :after copy-advice)
                             (unwind-protect
                                 (funcall callback resp info raw)
                               (advice-remove 'kill-new copy-advice)))))))
                (apply orig-fn prompt request-args)))))
    (advice-add 'gptel-request :around request-advice)
    (unwind-protect
        (tl/gptel-suffix-send args)
      (advice-remove 'gptel-request request-advice))))

** MCP
[[https://github.com/lizqwerscott/mcp.el][mcp.el]] is an Emacs *client* for the [[https://modelcontextprotocol.io][Model Context Protocol]] - a standard way for LLM tools to be supplied by standalone "MCP servers". With it,
=gptel= can call tools (read/write files, fetch URLs, query databases, hit APIs) provided by any MCP server you configure.

Below we register the official =filesystem= server (launched on demand via =npx=), scoped to my Emacs directory and home.
#+begin_src emacs-lisp
;; Install and configure servers
(use-package mcp
  :ensure t
  :after gptel                                  ; mcp is consumed by gptel here, so defer until it loads
  :custom
  (mcp-hub-servers
   `(("filesystem"                               ; stdio server, spawned by npx
      . (:command "npx"
         :args ("-y" "@modelcontextprotocol/server-filesystem")
         ;; `:roots' = directories the server may touch. mcp.el expands "~".
         :roots (,user-emacs-directory "~/")))
   ("fetch"                                      ; URL reader (HTML to markdown), spawned by uvx (needs uv installed)
      . (:command "uvx" :args ("mcp-server-fetch")))))
  :config
  (require 'mcp-hub)
  ;; Boot every configured server once gptel (and thus mcp) has loaded.
  ;; Idempotent: servers already running are skipped.
  (mcp-hub-start-all-server))

;; Bridge MCP tools into gptel
(with-eval-after-load 'gptel
  (require 'gptel-integrations)
  ;; Register every connected server's tools as gptel tools. On the first
  ;; gptel buffer this also starts any not-yet-running servers; later calls
  ;; are a no-op (tools are already registered globally).
  (add-hook 'gptel-mode-hook #'gptel-mcp-connect))
#+end_src

** Emacs MCP Server
Where the =mcp= package above is an MCP *client* (it consumes external servers from inside Emacs), =rhblind/emacs-mcp-server= is the inverse: an MCP *server* written in pure Elisp that runs inside this very Emacs process and exposes it over a Unix socket. External LLM clients — including the =oh-my-pi= coding agent — can then =eval-elisp=, read/write buffers, inspect diagnostics, and drive org-mode/org-roam against the *live* session.

It listens on a Unix domain socket bridged to clients via =socat= (already installed at =/usr/bin/socat=). We pin the socket to =~/.cache/emacs/emacs-mcp-server.sock= (matching this config's =user-emacs-directory=) and auto-start the server on launch.

** Agent-shell dictation (push-to-talk)
OMP's hold-=Space= speech-to-text is a TUI-only feature: it lives in the interactive editor's input loop (=onSpaceHoldStart= / =onSpaceHoldEnd=, gated by =SPACE_HOLD_RELEASE_MS=). agent-shell runs =omp acp= — the headless ACP protocol — which never starts the TUI, so the gesture has no code path to fire in the agent-shell buffer. This is a press-to-toggle alternative (the same shape as OMP's =app.stt.toggle= chord): it records via =pw-record= and transcribes with the same on-device Parakeet TDT model OMP's TUI uses, driven standalone through sherpa-onnx. The transcriber at =~/.config/.local/bin/omp-stt-transcribe= reuses the cached model at =~/.omp/agent/cache/tiny-models/csukuangfj/sherpa-onnx-nemo-parakeet-tdt-0.6b-v3-int8/=, so dictation matches the TUI experience. Install the runtime once with =uv tool install sherpa-onnx=. Bound to =C-c SPC= in =agent-shell-mode-map=.

#+begin_src emacs-lisp
(defvar tl/agent-shell-dictate--proc nil
  "Background `pw-record' process while dictating, or nil.")
(defvar tl/agent-shell-dictate--file "/tmp/omp-dictate.wav"
  "Temp WAV written by `pw-record' for dictation.")
(defvar tl/agent-shell-dictate--buffer nil
  "Buffer that started the current dictation recording.")

(defun tl/agent-shell-dictate ()
  "Press-to-talk speech-to-text for the agent-shell prompt.
Records 16 kHz mono with `pw-record', then transcribes with the
OMP-cached Parakeet model via sherpa-onnx and inserts the text at
point.  First call starts recording; second call stops and transcribes.
This mirrors OMP's `app.stt.toggle' press-to-toggle alternative, needed
because `omp acp' (agent-shell's backend) does not run the TUI STT path."
  (interactive)
  (if tl/agent-shell-dictate--proc
      (progn
        (message "transcribing…")
        (signal-process tl/agent-shell-dictate--proc 2)) ; SIGINT flushes the WAV header
    (setq tl/agent-shell-dictate--buffer (current-buffer))
    (let ((proc (start-process "omp-dictate" "*omp-dictate*"
                               "pw-record" "--rate" "16000" "--channels" "1"
                               tl/agent-shell-dictate--file)))
      (set-process-sentinel
       proc
       (lambda (p _event)
         (when (memq (process-status p) '(exit signal))
           (setq tl/agent-shell-dictate--proc nil)
           (tl/agent-shell-dictate--transcribe))))
      (setq tl/agent-shell-dictate--proc proc))
    (message "● REC — press again to transcribe")))

(defun tl/agent-shell-dictate--transcribe ()
  "Asynchronously transcribe the recording; insert the result at point."
  (let ((buf tl/agent-shell-dictate--buffer)
        (file tl/agent-shell-dictate--file))
    (if (not (file-exists-p file))
        (message "dictation: no recording file")
      (let ((script (expand-file-name "omp-stt-transcribe"
                                      "~/.config/.local/bin"))
            (accum ""))
        (let ((proc (make-process
                     :name "omp-stt"
                     :buffer nil
                     :command (list script file)
                     :noquery t)))
          (set-process-filter
           proc
           (lambda (_p out) (setq accum (concat accum out))))
          (set-process-sentinel
           proc
           (lambda (_p _event)
             (let ((text (string-trim accum)))
               (if (string-empty-p text)
                   (message "dictation: no speech detected")
                 (when (buffer-live-p buf)
                   (with-current-buffer buf
                     (insert text) (insert " ")))
                 (message "dictation: %s" text))))))))))

  (define-key agent-shell-mode-map (kbd "C-c SPC") #'tl/agent-shell-dictate))

** Agent-shell Bookmark

    (if (region-active-p)
        (setq xp1 (region-beginning) xp2 (region-end))
      (save-excursion
        (skip-chars-backward "[:alpha:]")
        (setq xp1 (point))
        (skip-chars-forward "[:alpha:]")
        (setq xp2 (point))))
    (when (not (eq last-command this-command))
      (put this-command 'state 0))
    (cond
     ((equal 0 (get this-command 'state))
      (upcase-initials-region xp1 xp2)
      (put this-command 'state 1))
     ((equal 1 (get this-command 'state))
      (upcase-region xp1 xp2)
      (put this-command 'state 2))
     ((equal 2 (get this-command 'state))
      (downcase-region xp1 xp2)
      (put this-command 'state 0)))))

(defun tl/org-roam-copy-finished-to-today ()
  "Copy the just-finished heading into today's roam daily note.
Works in both `org-mode' and `org-agenda-mode' buffers."
  (interactive)
  ;; ponytail: when fired from `org-after-todo-state-change-hook' via
  ;; `org-agenda-todo', the current buffer is the agenda (no file, no
  ;; subtree). `org-refile' then runs `org-copy-subtree' on agenda text
  ;; (a no-op) and `org-paste-subtree' on the stale kill ring, raising
  ;; "The kill is not a (set of) tree(s)". Resolve the agenda line's
  ;; heading marker and refile from the real org buffer instead.
  (let* ((src-marker (if (eq major-mode 'org-agenda-mode)
                         (org-get-at-bol 'org-hd-marker)
                       (point-marker)))
         (org-refile-keep t)
         (org-roam-dailies-capture-templates
          '(("t" "tasks" entry "%?"
             :if-new (file+head "%<%Y-%m-%d>.org" "#+title: %<%Y-%m-%d>\n"))))
         (org-after-refile-insert-hook #'save-buffer)
         (org-reverse-note-order t)
         today-file pos)
    (save-window-excursion
      (org-roam-dailies--capture (current-time) t)
      (setq today-file (buffer-file-name))
      (with-current-buffer (find-file-noselect today-file)
        (goto-char (point-min))
        (or (re-search-forward "^\\* Tasks" nil t)
            (progn
              (goto-char (point-min))
              (end-of-line)
              (insert "\n\n* Tasks")))
        (setq pos (point))))
    (when (and src-marker (marker-buffer src-marker))
      (with-current-buffer (marker-buffer src-marker)
        (save-excursion
          (goto-char src-marker)
          (unless (equal (file-truename today-file)
                         (file-truename (buffer-file-name)))
            (org-refile nil nil (list "Tasks" today-file nil pos))))))))

(add-to-list 'org-after-todo-state-change-hook
             (lambda ()
               (when (or (equal org-state "DONE")
                         (equal org-state "CNCL"))
                 (tl/org-roam-copy-finished-to-today))))

By default Org only exports the first three levels of headings as … headings. This is rather unfortunate as my documents frequently stray far beyond three levels of depth. The two main formats I care about exporting to are LaTeX and HTML. **** TOC as a collapsable tree
The TOC is much nicer to navigate as a collapsable tree. Unfortunately we cannot achieve this with CSS alone. Thankfully we can avoid JS though, by adapting the TOC generation code to use a ~label~ for each item, and a hidden ~checkbox~ to keep track of state.

To add this, we need to change one line in ~org-html--format-toc-headline~.

Since we can actually accomplish the desired effect by adding advice /around/ the function, without overriding it --- let's do that to reduce the bug surface of this config a tad.
#+begin_src emacs-lisp

**** Extra special strings
LaTeX already recognises =---= and =--= as em/en-dashes, =\-= as a shy hyphen, and the
conversion of =...= to =\ldots{}= is hardcoded into ~org-latex-plain-text~ (unlike
~org-html-plain-text~).

I'd quite like to also recognise =->= and =<-=, so let's set come up with some advice.

#+begin_src emacs-lisp
(defvar org-latex-extra-special-string-regexps
  '(("<->" . "\\\\(\\\\leftrightarrow{}\\\\)")
    ("->" . "\\\\textrightarrow{}")
    ("<-" . "\\\\textleftarrow{}")))

(defun org-latex-convert-extra-special-strings (string)
  "Convert special characters in STRING to LaTeX."
  (dolist (a org-latex-extra-special-string-regexps string)
    (let ((re (car a))
          (rpl (cdr a)))
      (setq string (replace-regexp-in-string re rpl string t)))))

(defun tl/org-latex-plain-text-extra-special-a (orig-fn text info)
  "Make `org-latex-plain-text' handle some extra special strings."
  (let ((output (funcall orig-fn text info)))
    (when (plist-get info :with-special-strings)
      (setq output (org-latex-convert-extra-special-strings output)))
    output))

(advice-add 'org-latex-plain-text :around #'tl/org-latex-plain-text-extra-special-a)
#+end_src

*** ASCII export
To start with, why settle for ASCII when UTF-8 exists?
#+begin_src emacs-lisp

#+end_src

The ASCII export is generally fairly nice. I think the main aspect that could benefit from improvement is the appearance of LaTeX fragments. There's a nice utility we can use to create unicode representation, which are much nicer. It's called ~latex2text~, and it's part of the =python-pylatexenc=.

#+begin_src emacs-lisp
(when (executable-find "latex2text")
  (eval-after-load 'ox-ascii
    '(progn
       (defvar org-ascii-convert-latex t
         "Use latex2text to convert LaTeX elements to unicode.")

       (defun tl/org-ascii-latex-environment-unicode (latex-environment _contents info)
         "Transcode a LATEX-ENVIRONMENT element from Org to ASCII, converting to unicode.
CONTENTS is nil.  INFO is a plist holding contextual information."
         (when (plist-get info :with-latex)
           (org-ascii--justify-element
            (org-remove-indentation
             (let* ((latex (org-element-property :value latex-environment))
                    (unicode (and (eq (plist-get info :ascii-charset) 'utf-8)
                                  org-ascii-convert-latex
                                  (call-process "latex2text" nil t nil "-q" "--code" latex))))
               (if (= (car unicode) 0) ; utf-8 set, and successfully ran latex2text
                   (cdr unicode) latex)))
            latex-environment info)))

       (advice-add 'org-ascii-latex-environment :override #'tl/org-ascii-latex-environment-unicode)

       (defun tl/org-ascii-latex-fragment-unicode (latex-fragment _contents info)
         "Transcode a LATEX-FRAGMENT object from Org to ASCII, converting to unicode.
CONTENTS is nil.  INFO is a plist holding contextual information."
         (when (plist-get info :with-latex)
           (let* ((latex (org-element-property :value latex-fragment))
                  (unicode (and (eq (plist-get info :ascii-charset) 'utf-8)
                                org-ascii-convert-latex
                                (call-process "latex2text" nil t nil "-q" "--code" latex))))
             (if (and unicode (= (car unicode) 0)) ; utf-8 set, and successfully ran latex2text
                 (cdr unicode) latex))))

       (advice-add 'org-ascii-latex-fragment :override #'tl/org-ascii-latex-fragment-unicode))))

* Performance Optimizations
Runtime tuning for this Emacs 30 native-comp build. The startup-critical opts
(GC deferral, =file-name-handler-alist= blanking, frame-inhibit, auto-mode
case-fold) live in =early-init.el= — they only work there, because this file is
*tangled then loaded* by =init.el= *after* =early-init.el= has already run.

The settings below tune *steady-state* behavior: they take effect once packages
are loaded and are safe to re-evaluate at any time.

#+begin_src emacs-lisp
;; --- Read subprocess output in 1 MB chunks --------------------------------
;; Emacs 30's default is 64 KB (older Emacsen shipped 4 KB). 1 MB cuts the
;; number of `read()' syscalls fired by process filters — helps `consult-ripgrep',
;; magit's git subprocess, compilation/shell buffers, and any future LSP.
;; Cheap and safe: a filter that reads less simply returns a shorter chunk.
(setq read-process-output-max (* 1024 1024))   ; 1 MB

;; --- Defer native-comp signature computation ------------------------------
;; This build reports `(native-comp-available-p)' => t. Deferring sign
;; compilation avoids blocking the async (.eln) compiler on certificate checks
;; and lets background compilation of installed packages proceed faster.
(setq comp-defer-sign-compilation t)

;; --- Survive minified / very-long-line files ------------------------------
;; `global-so-long-mode' (built in since Emacs 27) detects buffers whose lines
;; are too long for Emacs to handle briskly and mitigates them locally (disables
;; font-lock, line numbers, etc.). Without it, opening a multi-MB minified
;; JS/JSON/CSV can pin a core and freeze the frame for minutes.
(when (fboundp 'global-so-long-mode)
  (global-so-long-mode 1))

;; --- Defer font-lock in large buffers -------------------------------------
;; `locally-defer-font-lock' (defined under [[#org-mode][Org Mode]]) switches
;; jit-lock to deferred + stealth mode once a buffer exceeds 50 KB, so editing
;; big files stops feeling like "typing through molasses". Small buffers keep
;; instant font-lock (its `when' guard is a no-op below the threshold).
;; Activated here for *every* file / major-mode, not just org.
(add-hook 'find-file-hook #'locally-defer-font-lock)
(add-hook 'after-change-major-mode-hook #'locally-defer-font-lock)
#+end_src
